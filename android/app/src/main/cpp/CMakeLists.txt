cmake_minimum_required(VERSION 3.14)
project(driverloader LANGUAGES CXX C)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if (${CMAKE_ANDROID_ARCH_ABI} STREQUAL "arm64-v8a")
    add_subdirectory(adrenotools)
    add_library(driverloader SHARED driver_loader.cpp)
    target_include_directories(driverloader PRIVATE adrenotools/include)
    target_link_libraries(driverloader PRIVATE adrenotools log dl android)
else()
    add_library(driverloader SHARED driver_loader.cpp)
    target_link_libraries(driverloader PRIVATE log dl android)
endif()
